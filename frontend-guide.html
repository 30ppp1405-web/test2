پیام‌های دریافتی از Hub

نام متدپارامترهاتوضیح
GetBuysSnapshot records[] لیست کامل سفارش‌های خرید — فقط یک‌بار بعد از اتصال
GetSellsSnapshot records[] لیست کامل سفارش‌های فروش — فقط یک‌بار بعد از اتصال
GetBuys record, deltaType تغییر در سفارش‌های خرید (بعد از snapshot)
GetSells record, deltaType تغییر در سفارش‌های فروش (بعد از snapshot)
Heartbeat {exchange, orderType, hash} هر ۵ ثانیه — برای تشخیص از دست رفتن پیام

ساختار record

فیلدنوعتوضیح
inumberchecksum ردیف (برای تشخیص تغییر)
pdecimalقیمت — کلید اصلی
adecimalمجموع مقدار در این سطح قیمتی
vdecimalمجموع ارزش در این سطح قیمتی
enumberنوع بازار (exchange type)

مقادیر deltaType

مقدارناماقدام در فرانت
1Addردیف جدید به جدول اضافه کن + re-sort بر اساس قیمت
2Removeردیف را از جدول حذف کن
3Updateفقط مقدار a و v را بروزرسانی کن — نیازی به re-sort نیست

قوانین کلیدی

⚠ قانون اول: هرگز Delta را بدون داشتن Snapshot معتبر اعمال نکن. تا زمانی که GetBuysSnapshot و GetSellsSnapshot دریافت نشده، پیام‌های GetBuys و GetSells را نادیده بگیر.
⚠ قانون دوم: بعد از هر قطعی (reconnect)، state را پاک کن و منتظر Snapshot جدید بمان. هرگز state قدیمی را با delta جدید merge نکن.
ℹ قانون سوم: در نوع Update، قیمت تغییر نکرده — فقط مقدار و ارزش عوض شده. جدول را re-sort نکن تا flickering نداشته باشیم.

// ── مقادیر enum سمت سرور ──────────────────────────────────────────────
// این مقادیر باید با EnmOrderType در سرور تطابق داشته باشند
// برای یافتن مقادیر صحیح: console.log(msg.orderType) را در Heartbeat اجرا کن
const ORDER_TYPE = {
  BUY: 0,   // ← مقدار EnmOrderType.Buy.GetEnumValue() را اینجا بگذار
  SELL: 1   // ← مقدار EnmOrderType.Sell.GetEnumValue() را اینجا بگذار
}

const DELTA = { ADD: 1, REMOVE: 2, UPDATE: 3 }

بخش ۲ — State و Flash

src/composables/useOrderBook.js — بخش ۲
export function useOrderBook () {

  // ── State اصلی ────────────────────────────────────────────────────────
  // کلید Map = price (عدد اعشاری)، مقدار = record
  const _buys  = ref(new Map())
  const _sells = ref(new Map())

  // true بعد از دریافت اولین Snapshot — تا آن موقع delta اعمال نمی‌شود
  const ready  = ref(false)

  // وضعیت اتصال: 'connecting' | 'connected' | 'disconnected'
  const status = ref('disconnected')

  // Flash animations: کلید = 'buy-{price}' یا 'sell-{price}'، مقدار = 'add' | 'update'
  const flashes = ref({})

  // نوع بازار فعال (extype) — برای re-subscribe بعد از reconnect
  let currentExtype = 1
  let hub = null

  // ── Computed rows (مرتب‌شده برای نمایش) ─────────────────────────────
  // خرید: بالاترین قیمت اول
  const buyRows = computed(() =>
    [..._buys.value.values()].sort((a, b) => b.p - a.p)
  )

  // فروش: پایین‌ترین قیمت اول
  const sellRows = computed(() =>
    [..._sells.value.values()].sort((a, b) => a.p - b.p)
  )

  // ── Flash helpers ────────────────────────────────────────────────────
  function setFlash (side, price, type) {
    flashes.value = { ...flashes.value, [`${side}-${price}`]: type }
    // انیمیشن ۶۰۰ms طول می‌کشد
    setTimeout(() => {
      const f = { ...flashes.value }
      delete f[`${side}-${price}`]
      flashes.value = f
    }, 600)
  }

  function getFlash (side, price) {
    return flashes.value[`${side}-${price}`] || null
  }

بخش ۳ — Delta Apply و Hash

src/composables/useOrderBook.js — بخش ۳
  // ── اعمال تغییرات روی Map ────────────────────────────────────────────
  function applyDelta (mapRef, side, record, deltaType) {
    // کپی Map برای trigger کردن reactivity در Vue
    const map = new Map(mapRef.value)

    if (deltaType === DELTA.ADD) {
      // قیمت جدید وارد top-25 شد
      map.set(record.p, record)
      mapRef.value = map
      setFlash(side, record.p, 'add')

    } else if (deltaType === DELTA.REMOVE) {
      // قیمت از top-25 خارج شد
      map.delete(record.p)
      mapRef.value = map
      // flash لازم نیست، ردیف حذف می‌شود

    } else if (deltaType === DELTA.UPDATE) {
      // قیمت همان است، فقط مقدار/ارزش عوض شده
      // نکته: re-sort نمی‌کنیم چون قیمت (کلید مرتب‌سازی) تغییر نکرده
      if (map.has(record.p)) {
        map.set(record.p, record)
        mapRef.value = map
        setFlash(side, record.p, 'update')
      }
    }
  }

  // ── Hash Verification (Web Crypto API) ───────────────────────────────
  // الگوریتم باید دقیقاً با ComputeHash() سمت سرور تطابق داشته باشد:
  // input = "{price}:{checksum},{price}:{checksum},..." مرتب‌شده بر اساس price
  async function computeHash (map) {
    const input = [...map.entries()]
      .sort((a, b) => a[0] - b[0])
      .map(([k, v]) => `${k}:${v.i}`)
      .join(',')
    const buf = await crypto.subtle.digest(
      'SHA-256', new TextEncoder().encode(input))
    return [...new Uint8Array(buf)]
      .map(b => b.toString(16).padStart(2, '0'))
      .join('').slice(0, 8).toUpperCase()
  }

بخش ۴ — اتصال SignalR

src/composables/useOrderBook.js — بخش ۴
  // ── Subscribe به یک نوع بازار ────────────────────────────────────────
  // این متد را بعد از start() و هر بار که کاربر نوع بازار را عوض کرد صدا بزن
  function subscribe (extype) {
    currentExtype  = extype
    ready.value    = false
    _buys.value    = new Map()
    _sells.value   = new Map()
    hub?.invoke('Switch', extype).catch(console.error)
  }

  // ── برقراری اتصال ────────────────────────────────────────────────────
  async function start () {
    hub = new signalR.HubConnectionBuilder()
      .withUrl(HUB_URL)
      .withAutomaticReconnect([500, 1000, 2000, 5000, 10000])
      .configureLogging(signalR.LogLevel.Warning)
      .build()

    // ── Snapshot handlers ─────────────────────────────────────────────
    // جایگزینی کامل state — اولین پیام پس از اتصال
    hub.on('GetBuysSnapshot', records => {
      const map = new Map()
      records.forEach(r => map.set(r.p, r))
      _buys.value = map
      ready.value = true   // از اینجا به بعد delta اعمال می‌شود
    })

    hub.on('GetSellsSnapshot', records => {
      const map = new Map()
      records.forEach(r => map.set(r.p, r))
      _sells.value = map
    })

    // ── Delta handlers ────────────────────────────────────────────────
    // فقط بعد از دریافت Snapshot معتبر اعمال می‌شوند
    hub.on('GetBuys', (record, deltaType) => {
      if (!ready.value) return
      applyDelta(_buys, 'buy', record, deltaType)
    })

    hub.on('GetSells', (record, deltaType) => {
      if (!ready.value) return
      applyDelta(_sells, 'sell', record, deltaType)
    })

    // ── Heartbeat ─────────────────────────────────────────────────────
    // هر ۵ ثانیه: hash جدول خود را با سرور مقایسه کن
    // اگر مغایرت داشت: درخواست Snapshot جدید بده
    hub.on('Heartbeat', async msg => {
      if (!ready.value) return
      const map  = msg.orderType === ORDER_TYPE.BUY ? _buys.value : _sells.value
      const hash = await computeHash(map)
      if (hash !== msg.hash) {
        console.warn('[orderbook] hash mismatch — requesting snapshot')
        hub.invoke('GetSnapshot').catch(() => {})
      }
    })

    // ── Connection events ─────────────────────────────────────────────
    hub.onreconnecting(() => {
      status.value = 'connecting'
      ready.value  = false
      // state را پاک نمی‌کنیم تا UI خالی نشود، ولی delta قبول نمی‌کنیم
    })

    hub.onreconnected(() => {
      status.value = 'connected'
      // state پاک و Snapshot جدید درخواست می‌شود
      subscribe(currentExtype)
    })

    hub.onclose(() => {
      status.value = 'disconnected'
    })

    // ── شروع اتصال ───────────────────────────────────────────────────
    try {
      status.value = 'connecting'
      await hub.start()
      status.value = 'connected'
    } catch (e) {
      status.value = 'disconnected'
      throw e   // به کامپوننت فرستاده می‌شود تا نمایش خطا بدهد
    }
  }

  function stop () {
    hub?.stop()
  }

  // وقتی کامپوننت از DOM خارج شد، اتصال را ببند
  onUnmounted(stop)

  return {
    buyRows, sellRows,
    status, ready, flashes,
    getFlash, subscribe, start, stop
  }
}

src/pages/OrderBookPage.vue — <script setup>
<script setup>
import { ref, computed, onMounted } from 'vue'
import { useOrderBook } from 'src/composables/useOrderBook'

const {
  buyRows, sellRows,
  status, ready, getFlash,
  subscribe, start
} = useOrderBook()

// ── انتخاب نوع بازار ────────────────────────────────────────────────────
const selectedExtype = ref(7)          // مقدار پیش‌فرض
const extypeOptions  = [1, 2, 3, 7]   // لیست بازارهای موجود

function onExtypeChange (val) {
  lastUpdate.value = null
  subscribe(val)
}

// ── زمان آخرین بروزرسانی ───────────────────────────────────────────────
const lastUpdate = ref(null)

// هر بار که buyRows یا sellRows تغییر کرد، timestamp را بروزرسانی کن
// (این watcher فقط زمانی که ready=true فعال است)
import { watch } from 'vue'
watch([buyRows, sellRows], () => {
  if (ready.value) {
    lastUpdate.value = new Date().toLocaleTimeString('fa')
  }
})

// ── وضعیت اتصال ────────────────────────────────────────────────────────
const statusColor = computed(() => ({
  connected:    'positive',
  connecting:   'warning',
  disconnected: 'negative'
}[status.value]))

const statusText = computed(() => ({
  connected:    'متصل',
  connecting:   'در حال اتصال...',
  disconnected: 'قطع شده'
}[status.value]))

// ── Flash class برای row ────────────────────────────────────────────────
function getRowClass (side, price) {
  const f = getFlash(side, price)
  if (f === 'add')    return 'ob-flash-add'
  if (f === 'update') return 'ob-flash-update'
  return ''
}

// ── فرمت عدد ────────────────────────────────────────────────────────────
function fmt (n) {
  const num = Number(n)
  return num % 1 === 0
    ? num.toLocaleString('en')
    : num.toLocaleString('en', { minimumFractionDigits: 2, maximumFractionDigits: 8 })
}

// ── شروع اتصال در mount ─────────────────────────────────────────────────
onMounted(async () => {
  try {
    await start()
    subscribe(selectedExtype.value)
  } catch (e) {
    console.error('[OrderBook] Failed to connect:', e)
  }
})
</script>

عملکرد در حجم بالا

ℹ نکته Performance: در نوع UPDATE، Vue به خاطر computed فقط ردیف‌های تغییر یافته را re-render می‌کند. برای ۲۵ ردیف این کاملاً بهینه است. اگر در آینده تعداد ردیف‌ها زیاد شد، از v-memo یا VirtualScroll استفاده کن.

۱. مطمئن شو سرور در حال اجرا است
۲. بررسی کن که AllowCredentials() در Startup سرور وجود داشته باشد
۳. اگر از vite dev server استفاده می‌کنی، proxy تنظیم کن:
quasar.config.js — proxy برای حل CORS در development
devServer: {
  proxy: {
    '/OrderHub': {
      target: 'http://localhost:25814',
      ws: true,       // ← مهم: برای WebSocket
      changeOrigin: true
    }
  }
}

اگر proxy استفاده کردی، HUB_URL در composable را به '/OrderHub' تغییر بده.

جریان Reconnect:

۱. Hub قطع می‌شود → onreconnecting → ready = false
۲. دیگر delta اعمال نمی‌شود
۳. UI همان داده قبلی را نشان می‌دهد (نه خالی)
۴. اتصال برقرار می‌شود → onreconnected → subscribe()
۵. سرور Snapshot کامل می‌فرستد → state جایگزین می‌شود → ready = true
